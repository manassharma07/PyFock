pyfock.ase_calculator

ASE calculator for PyFock.

This module provides a subprocess-backed ASE calculator that writes a standard PyFock API script into a working directory, executes it in the background, and parses results from the saved output file.

 35def check_convergence(output_file):
 36    """
 37    Check whether the PyFock calculation converged from its output file.
 38
 39    Returns
 40    -------
 41    tuple
 42        ``(converged, message)`` where ``message`` is a short matching line if
 43        convergence failed.
 44    """
 45
 46    failure_patterns = [
 47        "SCF NOT Converged",
 48        "NOT CONVERGED",
 49        "did not converge",
 50        "convergence not achieved",
 51        "ERROR:",
 52    ]
 53    success_patterns = [
 54        "SCF Converged after",
 55        "SCF CONVERGED",
 56    ]
 57
 58    try:
 59        with open(output_file, "r", encoding="utf-8") as handle:
 60            content = handle.read()
 61    except FileNotFoundError:
 62        return False, f"Output file '{output_file}' not found"
 63
 64    content_lower = content.lower()
 65    for pattern in failure_patterns:
 66        if pattern.lower() in content_lower:
 67            for line in content.splitlines():
 68                if pattern.lower() in line.lower():
 69                    return False, line.strip()
 70            return False, pattern
 71
 72    for pattern in success_patterns:
 73        if pattern.lower() in content_lower:
 74            return True, None
 75
 76    return False, "Could not determine convergence from PyFock output"
 77

 79def _parse_result_marker(output_file):
 80    marker = "PYFOCK_RESULT_JSON="
 81    with open(output_file, "r", encoding="utf-8") as handle:
 82        for line in handle:
 83            if line.startswith(marker):
 84                return json.loads(line[len(marker) :].strip())
 85    raise RuntimeError(
 86        f"Could not find '{marker}' marker in '{output_file}'. "
 87        "Check the output file for PyFock errors."
 88    )

 93    ASE calculator wrapper for PyFock DFT/HF calculations.

168    def _dft_attribute_names(cls):
169        if cls._cached_dft_attr_names is None:
170            mol = Mol(atoms=[["H", 0.0, 0.0, 0.0], ["H", 0.0, 0.0, 0.74]])
171            dft_obj = DFT(mol, mol.basis, xc="PBE")

175    def _canonicalize_options(self, kwargs):
176        options = dict(kwargs)
177        if "functional" in options:
178            if "xc" in options:
179                raise TypeError("Use either 'functional' or 'xc', not both.")
180            options["xc"] = options.pop("functional")
181        if "DF" in options:
182            if "isDF" in options:
183                raise TypeError("Use either 'DF' or 'isDF', not both.")
184            options["isDF"] = options.pop("DF")
185        return options

187    def _validate_option_names(self, options):
188        allowed = self._dft_attribute_names()
189        unknown = sorted(set(options) - allowed)
190        if unknown:
191            raise TypeError(
192                "Unknown PyFock calculator option(s): " + ", ".join(unknown)
193            )

195    def _default_basis_name(self, atoms):
196        if not np.any(atoms.pbc):
197            return "def2-SVP"
198        return "def2-SVP"

200    def _state_token(self, atoms):
201        payload = {
202            "symbols": atoms.get_chemical_symbols(),
203            "positions": np.asarray(atoms.get_positions(), dtype=np.float64).round(12).tolist(),
204            "cell": np.asarray(atoms.get_cell().array, dtype=np.float64).round(12).tolist(),
205            "pbc": [bool(x) for x in atoms.pbc],
206            "charge": self.parameters["charge"],
207            "basis": self.parameters["basis"],
208            "auxbasis": self.parameters["auxbasis"],
209            "options": self.pyfock_options,
210        }
211        encoded = json.dumps(payload, sort_keys=True).encode("utf-8")
212        return hashlib.sha256(encoded).hexdigest()

214    def _to_ev_forces(self, forces_au_bohr):
215        factor = Data.au2eVFactor / Data.Bohr2AngsFactor
216        return np.asarray(forces_au_bohr, dtype=np.float64) * factor

218    def _to_eang_dipole(self, dipole_au):
219        return np.asarray(dipole_au, dtype=np.float64) * Data.Bohr2AngsFactor

221    def _compute_dispersion_correction(self, atoms, compute_forces):
222        try:
223            from torch_dftd.torch_dftd3_calculator import TorchDFTD3Calculator
224        except ImportError as exc:
225            raise ImportError(
226                "Dispersion correction requires the optional 'torch-dftd' package. "
227                "Install it with: pip install torch-dftd"
228            ) from exc
229
230        dispersion_kwargs = self.parameters.get("dispersion_kwargs")
231        if dispersion_kwargs is None:
232            dispersion_kwargs = {}
233        else:
234            dispersion_kwargs = dict(dispersion_kwargs)
235
236        dispersion_kwargs.setdefault("atoms", atoms.copy())
237        disp_atoms = dispersion_kwargs["atoms"]
238        disp_calc = TorchDFTD3Calculator(**dispersion_kwargs)
239        disp_atoms.calc = disp_calc
240
241        disp_energy = float(disp_atoms.get_potential_energy())
242        disp_forces = None
243        if compute_forces:
244            disp_forces = np.asarray(disp_atoms.get_forces(), dtype=np.float64)
245        return disp_energy, disp_forces

247    def _next_step_dir(self):
248        self._iteration += 1
249        step_dir = os.path.join(self.directory, f"step_{self._iteration:04d}")
250        os.makedirs(step_dir, exist_ok=True)
251        self._last_step_dir = step_dir
252        return step_dir

254    def _get_workdir_for_state(self, atoms):
255        state_token = self._state_token(atoms)
256        if self._last_step_token == state_token and self._last_step_dir is not None:
257            os.makedirs(self._last_step_dir, exist_ok=True)
258            return state_token, self._last_step_dir
259
260        step_dir = self._next_step_dir()
261        self._last_step_token = state_token
262        return state_token, step_dir

264    def _write_xyz(self, atoms, filepath):
265        positions = atoms.get_positions()
266        with open(filepath, "w", encoding="utf-8") as handle:
267            handle.write(f"{len(atoms)}\n")
268            handle.write("Generated by PyFockCalculator\n")
269            for symbol, coord in zip(atoms.get_chemical_symbols(), positions):
270                handle.write(
271                    f"{symbol} {coord[0]:.16f} {coord[1]:.16f} {coord[2]:.16f}\n"

277    def _prepare_runtime_options(self):
278        options = dict(self.pyfock_options)
279        xc_value = options.get("xc")
280        user_set_df = "isDF" in options
281        user_set_rys = "rys" in options
282        user_set_direct_scf = "direct_scf" in options
283
284        if xc_value == "HF":
285            if user_set_df and options.get("isDF", True):
286                raise ValueError("PyFock HF through the DFT module requires DF=False.")
287            if not user_set_df:
288                options["isDF"] = False
289            if user_set_rys and options.get("rys", True):
290                raise ValueError("PyFock HF currently requires rys=False.")
291            if not user_set_rys:
292                options["rys"] = False
293            if user_set_direct_scf and not options.get("direct_scf", False):
294                raise ValueError("PyFock HF currently requires direct_scf=True.")
295            if not user_set_direct_scf:
296                options["direct_scf"] = True
297
298        return options

300    def _write_run_script(self, atoms, workdir, task_name, compute_forces=False, compute_dipole=False):
301        options = self._prepare_runtime_options()
302        basis_name = self.parameters["basis"] or self._default_basis_name(atoms)
303        auxbasis_name = self.parameters["auxbasis"] or "def2-universal-jfit"
304        xyz_filename = "structure.xyz"
305        output_filename = f"output_pyfock_{task_name}.txt"
306        script_path = os.path.join(workdir, f"run_pyfock_{task_name}.py")
307
308        option_lines = []
309        for key, value in sorted(options.items()):
310            option_lines.append(f"dft_obj.{key} = {self._render_value(value)}")
311        option_block = "\n".join(option_lines)
312
313        script = f"""import json
314import os
315import numpy as np

317from pyfock import Basis
318from pyfock import DFT
319from pyfock import DFT_NumGrad
320from pyfock import Integrals

325def compute_homo_lumo_gap(dft_obj):
326    eigvalues = getattr(dft_obj, "mo_energies", None)
327    occupations = getattr(dft_obj, "mo_occupations", None)
328    if eigvalues is None or occupations is None:
329        return None, None
330    eigvalues = np.asarray(eigvalues)
331    occupations = np.asarray(occupations)
332    occupied = np.where(occupations > 1e-8)[0]
333    if len(occupied) == 0 or occupied[-1] + 1 >= len(eigvalues):
334        return None, None
335    homo_idx = occupied[-1]
336    lumo_idx = homo_idx + 1
337    gap_au = float(eigvalues[lumo_idx] - eigvalues[homo_idx])
338    gap_ev = float(gap_au * Data.au2eVFactor)
339    return gap_au, gap_ev
340
341
342ncores = {self._render_value(options.get("ncores", 1))}
343if ncores is not None:
344    os.environ["OMP_NUM_THREADS"] = str(ncores)
345    os.environ["OPENBLAS_NUM_THREADS"] = str(ncores)
346    os.environ["MKL_NUM_THREADS"] = str(ncores)
347    os.environ["VECLIB_MAXIMUM_THREADS"] = str(ncores)
348    os.environ["NUMEXPR_NUM_THREADS"] = str(ncores)
349
350mol = Mol(coordfile={self._render_value(xyz_filename)}, charge={self._render_value(self.parameters["charge"])})
351basis = Basis(mol, {{"all": Basis.load(mol=mol, basis_name={self._render_value(basis_name)})}})
352
353use_df = {self._render_value(options.get("isDF", True))}
354if use_df:
355    auxbasis = Basis(mol, {{"all": Basis.load(mol=mol, basis_name={self._render_value(auxbasis_name)})}})
356else:
357    auxbasis = None
358
359dft_obj = DFT(mol, basis, auxbasis)
360{option_block}
361
362energy_au, dmat = dft_obj.scf()
363gap_au, gap_ev = compute_homo_lumo_gap(dft_obj)
364
365result = {{
366    "converged": bool(getattr(dft_obj, "converged", False)),
367    "niter": int(getattr(dft_obj, "niter", 0)),
368    "total_energy_au": float(energy_au),
369    "total_energy_ev": float(energy_au * Data.au2eVFactor),
370    "xc_energy_au": None if getattr(dft_obj, "XC_energy", None) is None else float(dft_obj.XC_energy),
371    "coulomb_energy_au": None if getattr(dft_obj, "J_energy", None) is None else float(dft_obj.J_energy),
372    "kinetic_energy_au": None if getattr(dft_obj, "Kinetic_energy", None) is None else float(dft_obj.Kinetic_energy),
373    "electron_nuclear_energy_au": None if getattr(dft_obj, "Nuc_energy", None) is None else float(dft_obj.Nuc_energy),
374    "nuclear_repulsion_energy_au": None if getattr(dft_obj, "Nuclear_repulsion_energy", None) is None else float(dft_obj.Nuclear_repulsion_energy),
375    "homo_lumo_gap_au": gap_au,
376    "homo_lumo_gap_ev": gap_ev,
377}}
378
379if {self._render_value(compute_forces)}:
380    grad_obj = DFT_NumGrad(
381        dft_obj,
382        step_size={self._render_value(self.parameters["force_step_size"])},
383        step_unit={self._render_value(self.parameters["force_step_unit"])},
384        method={self._render_value(self.parameters["force_method"])},
385        use_fixed_grids={self._render_value(self.parameters["force_use_fixed_grids"])},
386        verbose=False,
387    )
388    force_results = grad_obj.calculate()
389    result["forces_au_bohr"] = np.asarray(force_results["forces"]).tolist()
390
391if {self._render_value(compute_dipole)}:
392    dipole_matrix = Integrals.dipole_moment_mat_symm(basis)
393    dipole_au = mol.get_dipole_moment(dipole_matrix, dmat)
394    result["dipole_au"] = np.asarray(dipole_au).tolist()
395
396print("PYFOCK_RESULT_JSON=" + json.dumps(result, sort_keys=True))
397"""
398
399        with open(script_path, "w", encoding="utf-8") as handle:
400            handle.write(script)
401
402        return script_path, os.path.join(workdir, output_filename)

404    def _run_pyfock_script(self, workdir, script_path, output_path):
405        with open(output_path, "w", encoding="utf-8") as output_handle:
406            result = subprocess.run(
407                [sys.executable, os.path.basename(script_path)],
408                cwd=workdir,
409                stdout=output_handle,
410                stderr=subprocess.STDOUT,
411            )
412
413        if result.returncode != 0:
414            raise RuntimeError(
415                f"PyFock subprocess failed with exit code {result.returncode}. "
416                f"Check '{output_path}' for details."
417            )
418
419        convergence_mode = self.parameters["convergence_check"]
420        converged, message = check_convergence(output_path)
421        self.converged = converged
422        if convergence_mode != "ignore" and not converged:
423            error_message = (
424                f"PyFock calculation did not converge. Details: {message}. "
425                f"Check '{output_path}' for more information."
426            )
427            if convergence_mode == "error":
428                raise PyFockConvergenceError(error_message)
429            warnings.warn(error_message, PyFockConvergenceWarning)
430
431        summary = _parse_result_marker(output_path)
432        if convergence_mode == "ignore":
433            self.converged = bool(summary.get("converged", True))
434        return summary

457    def calculate(self, atoms=None, properties=None, system_changes=all_changes):
458        if properties is None:
459            properties = self.implemented_properties
460
461        super().calculate(atoms, properties, system_changes)
462
463        os.makedirs(self.directory, exist_ok=True)
464        state_token, step_dir = self._get_workdir_for_state(self.atoms)
465        xyz_path = os.path.join(step_dir, "structure.xyz")
466        self._write_xyz(self.atoms, xyz_path)
467
468        compute_forces = "forces" in properties
469        script_path, output_path = self._write_run_script(
470            self.atoms,
471            step_dir,
472            task_name="singlepoint",
473            compute_forces=compute_forces,
474            compute_dipole=False,
475        )
476        summary = self._run_pyfock_script(step_dir, script_path, output_path)
477        self._populate_common_results(summary)
478
479        if compute_forces:
480            if "forces_au_bohr" not in summary:
481                raise RuntimeError(
482                    f"Forces were requested but not found in '{output_path}'."
483                )
484            self.results["forces"] = self._to_ev_forces(summary["forces_au_bohr"])
485
486        self.pyfock_results["base_energy_ev"] = float(self.results["energy"])
487        self.pyfock_results["base_free_energy_ev"] = float(self.results["free_energy"])
488        if compute_forces:
489            self.pyfock_results["base_forces_ev_ang"] = np.asarray(
490                self.results["forces"], dtype=np.float64
491            ).tolist()
492
493        if self.parameters["dispersion"]:
494            disp_energy, disp_forces = self._compute_dispersion_correction(
495                self.atoms, compute_forces
496            )
497            self.results["energy"] += disp_energy
498            self.results["free_energy"] = self.results["energy"]
499            self.pyfock_results["dispersion_energy_ev"] = disp_energy
500            self.pyfock_results["total_energy_ev"] = float(self.results["energy"])
501            if compute_forces:
502                self.results["forces"] = self.results["forces"] + disp_forces
503                self.pyfock_results["dispersion_forces_ev_ang"] = disp_forces.tolist()
504                self.pyfock_results["total_forces_ev_ang"] = np.asarray(
505                    self.results["forces"], dtype=np.float64
506                ).tolist()
507
508        self._last_energy_token = state_token
509        self._last_dipole_token = None
510        self._last_dipole_eang = None

518    def get_dipole_moment(self, atoms=None):
519        if atoms is None:
520            atoms = self.atoms
521
522        state_token = self._state_token(atoms)
523        if self._last_energy_token != state_token or self._last_step_dir is None:
524            raise RuntimeError(
525                "Dipole moment is available only after get_potential_energy() "
526                "has been called for the current structure."
527            )
528
529        if self._last_dipole_token == state_token and self._last_dipole_eang is not None:
530            return self._last_dipole_eang.copy()
531
532        xyz_path = os.path.join(self._last_step_dir, "structure.xyz")
533        if not os.path.exists(xyz_path):
534            self._write_xyz(atoms, xyz_path)
535
536        script_path, output_path = self._write_run_script(
537            atoms,
538            self._last_step_dir,
539            task_name="dipole",
540            compute_forces=False,
541            compute_dipole=True,
542        )
543        summary = self._run_pyfock_script(self._last_step_dir, script_path, output_path)
544        dipole_au = summary.get("dipole_au")
545        if dipole_au is None:
546            raise RuntimeError(
547                f"Dipole moment was requested but not found in '{output_path}'."
548            )
549
550        dipole_eang = self._to_eang_dipole(dipole_au)
551        self._last_dipole_eang = dipole_eang
552        self._last_dipole_token = state_token
553        self.pyfock_results["dipole_au"] = dipole_au
554        self.pyfock_results["dipole_eang"] = dipole_eang.tolist()
555        return dipole_eang.copy()

557    def get_homo_lumo_gap(self, atoms=None, unit="eV"):
558        if atoms is None:
559            atoms = self.atoms
560
561        state_token = self._state_token(atoms)
562        if self._last_energy_token != state_token:
563            raise RuntimeError(
564                "HOMO-LUMO gap is available only after get_potential_energy() "
565                "has been called for the current structure."
566            )
567
568        if unit.lower() == "ev":
569            return self._last_homo_lumo_gap_ev
